2. CORE SPATIAL AUDIO CONCEPTS

2.1 Object-Based vs Channel-Based Audio

Traditional Channel-Based (5.1, 7.1 Surround)

Fixed speaker layout (Front L/C/R, Surround L/R, LFE)

Audio mixed to specific channels

Not scalable—different playback systems need different mixes

Object-Based (Spatial Audio)

Audio treated as independent objects with metadata

Each object has: audio data + position (X, Y, Z) + size + dynamics

Dynamically rendered to ANY speaker configuration

Scalable from headphones to cinema systems

Def spherical_to_cartesian(azimuth_deg, elevation_deg, distance_m):

  “””

  Convert spherical coordinates to 3D Cartesian space



  Spherical Coordinates:

  - Azimuth: -180° (left) to +180° (right), 0° = center front

  - Elevation: -90° (below) to +90° (above), 0° = ear level

  - Distance: meters from listener



  Cartesian Output:

  - X: Left (-) to Right (+)

  - Y: Below (-) to Above (+)

  - Z: Behind (-) to Front (+)

  “””



  Azimuth_rad = np.radians(azimuth_deg)

  Elevation_rad = np.radians(elevation_deg)



  # Spherical to Cartesian conversion

  X = distance_m * np.cos(elevation_rad) * np.sin(azimuth_rad)

  Y = distance_m * np.sin(elevation_rad)

  Z = distance_m * np.cos(elevation_rad) * np.cos(azimuth_rad)



  Return x, y, z
Def cartesian_to_spherical(x, y, z):

  “””Reverse conversion: Cartesian to spherical”””



  Distance = np.sqrt(x**2 + y**2 + z**2)



  # Azimuth: angle in XZ plane

  Azimuth_rad = np.arctan2(x, z)

  Azimuth_deg = np.degrees(azimuth_rad)



  # Elevation: angle above/below horizon

  Elevation_rad = np.arcsin(y / (distance + 1e-10))

  Elevation_deg = np.degrees(elevation_rad)



  Return azimuth_deg, elevation_deg, distance

4.1 Adaptive Rendering Algorithm

Python



Def adaptive_render(audio_object, speaker_config, listener_position):

  “””

  Calculate optimal speaker assignment for audio object

  Renders object to whatever speakers are available



  Algorithm:

  1. Get object position (relative to listener)

  2. Calculate angular distance to each speaker

  3. Assign gain based on proximity

  4. Normalize gains to preserve loudness
“””



# Get object position relative to listener

Obj_x, obj_y, obj_z = listener_position.get_object_relative_position(

    Audio_object.position_x,

    Audio_object.position_y,

    Audio_object.position_z

)



# Convert to spherical for easier angle calculations

Obj_azimuth, obj_elevation, obj_distance = cartesian_to_spherical(obj_x, obj_y, obj_z)



# Initialize speaker assignments

Speaker_assignments = {}



# Calculate gain for each available speaker

For speaker_name, speaker_pos in speaker_config.items():

    If speaker_name == “height_speakers” or speaker_name == “total_speakers”:

        Continue # Skip metadata fields



    # Angular distance from object to speaker

    Angle_diff = calculate_angular_distance(

        Obj_azimuth, obj_elevation,

        Speaker_pos[“azimuth”], speaker_pos[“elevation”]

    )



    # Only assign to speakers reasonably close to object

    If angle_diff < 90: # Within 90° cone
        # Gain based on angular distance (inverse square of angle)

        # Closer angles = higher gain

        Gain_linear = 1.0 / (1.0 + angle_diff)



        # Apply distance attenuation (inverse square law)

        Distance_factor = speaker_pos[“distance”] / (obj_distance + 0.1)

        Gain_linear *= distance_factor ** 0.5



        Speaker_assignments[speaker_name] = max(0, gain_linear)



  # Normalize gains to sum to 1.0 (preserve loudness)

  Total_gain = sum(speaker_assignments.values())

  If total_gain > 1e-10:

    For speaker in speaker_assignments:

        Speaker_assignments[speaker] /= total_gain



  Return speaker_assignments




Def calculate_angular_distance(azimuth1, elevation1, azimuth2, elevation2):

  “””

  Calculate angle between two directions in 3D space

  Uses spherical distance formula

  “””



  Az1_rad = np.radians(azimuth1)

  El1_rad = np.radians(elevation1)

  Az2_rad = np.radians(azimuth2)

  El2_rad = np.radians(elevation2)
  # Angular distance using spherical law of cosines

  Cos_distance = (

      Np.sin(el1_rad) * np.sin(el2_rad) +

      Np.cos(el1_rad) * np.cos(el2_rad) * np.cos(az2_rad – az1_rad)

  )



  # Clamp to avoid numerical errors

  Cos_distance = np.clip(cos_distance, -1.0, 1.0)



  Distance_rad = np.arccos(cos_distance)

  Distance_deg = np.degrees(distance_rad)



  Return distance_deg

Def design_parametric_eq(center_freq, gain_db, q_factor):

  “””Design peaking EQ filter”””

  W0 = 2 * np.pi * center_freq / 48000.0 # Normalized frequency



  Alpha = np.sin(w0) / (2 * q_factor)
  A = 10 ** (gain_db / 40.0)

Def demonstrate_hrtf_mechanism():

# Interaural Level Difference: Volume difference encodes azimuth

# Head shadow causes high frequencies on far side to be quieter

Def distance_reverb_model(distance_meters, room_size=”medium”):

  “””

  Closer objects = less reverb

  Distant objects = more reverb and reflections



  This simulates acoustic environment response

  “””



  # Reverb intensity increases with distance

      Reverb_wet_amount = 0.05 # Close-miked (minimal reverb)

  Elif distance_meters < 3:

      Reverb_wet_amount = 0.15 # Near field

  Elif distance_meters < 8:

      Reverb_wet_amount = 0.35 # Mid field

  Else:

      Reverb_wet_amount = 0.60 # Far field (very ambient)
  # Early reflection delay based on distance

  # Reflections arrive proportional to distance

  Early_reflection_delay_ms = distance_meters * 2.9 # Speed of sound



  # Reverb decay time increases with distance

  If room_size == “small”:

      Decay_time_base = 0.5

  Elif room_size == “medium”:

      Decay_time_base = 1.2

  Else: # large

      Decay_time_base = 2.0



  Decay_time_sec = decay_time_base + (distance_meters / 10.0)



  Return {

      “reverb_wet”: reverb_wet_amount,

      “early_delay_ms”: early_reflection_delay_ms,

      “decay_time_sec”: decay_time_sec,

      “predelay_ms”: 10, # Time before reverb tail starts

  }




Def apply_distance_reverb(audio, distance_m, room_size=”medium”):

  “””Apply combined distance processing: attenuation + reverb”””



  # Get distance parameters

  Distance_atten = distance_attenuation(distance_m)

  Absorption_filter = air_absorption_filter(distance_m)
  Reverb_config = distance_reverb_model(distance_m, room_size)



  # Apply attenuation

  Audio_attenuated = audio * distance_atten[“gain_linear”]



  # Apply high-frequency roll-off (air absorption)

  Audio_filtered = scipy.signal.lfilter(

      Absorption_filter[‘b’],

      Absorption_filter[‘a’],

      Audio_attenuated

  )



  # Apply reverb

  # (Simplified: use basic delay-based reverb)

  Reverb_signal = create_simple_reverb(

      Audio_filtered,

      Delay_ms=reverb_config[“early_delay_ms”],

      Decay_time=reverb_config[“decay_time_sec”]

  )



  # Mix dry and wet

  Output = (audio_filtered * (1 – reverb_config[“reverb_wet”]) +

         Reverb_signal * reverb_config[“reverb_wet”])



  Return output

    Times = [k[0] for k in self.keyframes]

    Ys = [k[2] for k in self.keyframes]

    Zs = [k[3] for k in self.keyframes]

  Def get_velocity(self, time_sec):

    “””Calculate velocity for Doppler effect”””



    Dt = 0.001 # 1ms time step



    X1, y1, z1 = self.get_position(time_sec – dt/2)

    X2, y2, z2 = self.get_position(time_sec + dt/2)



    # Velocity = change in position / time

    Vx = (x2 – x1) / dt

    Vy = (y2 – y1) / dt

    Vz = (z2 – z1) / dt



    # Velocity magnitude

    Speed = np.sqrt(vx**2 + vy**2 + vz**2)



    Return vx, vy, vz, speed

8.2 Doppler Pitch Shift
Python



Def calculate_doppler_shift(velocity_ms, source_frequency_hz,

                Speed_of_sound=343.0):

  “””

  Calculate pitch shift due to object movement



  Approaching object = higher frequency (pitch rises)

  Receding object = lower frequency (pitch drops)



  Formula: f’ = f * (c – v_observer) / (c – v_source)

  Simplified for stationary observer: f’ = f / (1 – v/c)

  “””



  # Doppler factor

  # Positive velocity = approaching = factor < 1 = higher pitch

  # Negative velocity = receding = factor > 1 = lower pitch



  Doppler_factor = 1.0 – (velocity_ms / speed_of_sound)



  # Shifted frequency

  Shifted_frequency = source_frequency_hz / doppler_factor



  # Pitch shift in cents (100 cents = 1 semitone)

  Cents_shift = 1200 * np.log2(doppler_factor)



  Return {

    “shifted_frequency”: shifted_frequency,

    “cents_shift”: cents_shift,
      “doppler_factor”: doppler_factor,

      “velocity_ms”: velocity_ms

  }

Def smooth_gain_changes(gain_envelope, smoothing_ms=5, sample_rate=48000):

  “””

  Smooth gain automation to prevent audible “zipper” artifacts



  Rapid gain changes at different sample values create clicks

  Solution: Apply moving average filter

  “””



  # Convert milliseconds to samples

Def detect_audible_artifacts(audio, sample_rate=48000):

  “””



  Artifacts_detected = {

      “zipper_noise”: False,

      “phase_cancellation”: False,

      “aliasing”: False,

      “clipping”: False,

      “pump_breathing”: False,

  }
  # 1. Zipper noise: rapid gain changes

    Artifacts_detected[“zipper_noise”] = True



  # 2. Phase cancellation: notches in frequency response

  If has_phase_issues(audio):

    Artifacts_detected[“phase_cancellation”] = True



  # 3. Aliasing: high-frequency artifacts

Def has_zipper_noise(audio, threshold=0.001):

  “””Detect rapid gain changes”””



  # Calculate frame-to-frame level changes
  Rms_envelope = calculate_rms_envelope(audio, frame_size=256)

  Differences = np.abs(np.diff(rms_envelope))



  # Excessive variation indicates zipper

  Mean_diff = np.mean(differences)

  Std_diff = np.std(differences)



  Has_artifacts = (std_diff / (mean_diff + 1e-10)) > threshold

  Return has_artifacts




Def has_aliasing(audio, sample_rate, frequency_threshold=None):

  “””Detect aliasing in high frequencies”””



  If frequency_threshold is None:

      Frequency_threshold = sample_rate / 4



  # FFT analysis

  Fft = np.fft.rfft(audio)

  Frequencies = np.fft.rfftfreq(len(audio), 1/sample_rate)

  Magnitude = np.abs(fft)



  # Check for energy above Nyquist/2

  Nyquist_idx = len(frequencies) // 2



  Energy_above_threshold = np.sum(

      Magnitude[frequencies > frequency_threshold]

  )

  Total_energy = np.sum(magnitude)
  Aliasing_ratio = energy_above_threshold / (total_energy + 1e-10)



  # More than 5% energy in aliasing region suggests problem

  Return aliasing_ratio > 0.05

A.1 KEY FORMULAS

Distance Attenuation (Inverse Square Law)

Text



Gain (dB) = 20 * log10(reference_distance / actual_distance)

Doppler Pitch Shift

Text



Shifted_frequency = original_frequency / (1 – velocity / speed_of_sound)

Pitch_shift (cents) = 1200 * log2(doppler_factor)

Interaural Time Difference

Text



ITD (seconds) = (head_width / speed_of_sound) * sin(azimuth)

Typical: ±0.7ms for ±90° azimuth

Angular Distance (Spherical)
Text



Cos(distance) = sin(el1) * sin(el2) + cos(el1) * cos(el2) * cos(az2 – az1)

Distance (degrees) = arccos(cos_distance)

Vector Panning Gain

Text



Gain = max(0, dot_product(pan_vector, speaker_vector))^1.5

Soft-Knee Compression

Text



If input < lower_knee:

  Output = input

Else if input > upper_knee:

  Excess = input – threshold

  Output = input – excess * (1 – 1/ratio)

Else (in knee):

  Knee_position = (input – lower_knee) / knee_width

  Compression_amount = excess * knee_position^2 * (1 – 1/ratio)

  Output = input – compression_amount

Def pan_to_speakers(azimuth, speaker_config):

  Gains = {}

  Azimuth_rad = np.radians(azimuth)



  For speaker_name, (speaker_az, _, _) in speaker_config.items():

      Speaker_az_rad = np.radians(speaker_az)



      # Angular distance

      Angle_diff = abs(azimuth_rad – speaker_az_rad)



      # Gain based on proximity

      Gain = max(0, 1.0 – angle_diff / np.pi)

      Gains[speaker_name] = gain



  # Normalize

  Total = sum(gains.values())

  For speaker in gains:

      Gains[speaker] /= (total + 1e-10)



  Return gains
# 4. Main rendering

Def render_to_speakers(audio_object, speaker_config):

  # Get speaker gains for object’s azimuth

  Gains = pan_to_speakers(audio_object.azimuth, speaker_config)



  # Distance attenuation

  Distance_gain = 1.0 / (audio_object.distance ** 2)



  # Apply to each speaker

  Output_speakers = {}

  For speaker, gain in gains.items():

    Output_speakers[speaker] = audio_object.audio * gain * distance_gain



  Return output_speakers
